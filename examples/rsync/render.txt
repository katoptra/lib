task: [label] python3 -I - /work/examples/rsync/staging $'application/x-gzip .gz .tgz .taz .tz .svgz\napplication/x-bzip2 .bz2 .tbz2 .tbz\napplication/x-xz .xz .txz\napplication/x-compress .Z\napplication/zstd .zst .tzst\napplication/x-lzip .lz\napplication/x-lzma .lzma' > /work/examples/rsync/.run/labels.raw <<'PY'
import mimetypes, os, sys

root, table = sys.argv[1], sys.argv[2]
endings = {s: f[0] for f in map(str.split, table.splitlines()) if f for s in f[1:]}
signatures = ((b"\x1f\x8b", "application/x-gzip"), (b"BZh", "application/x-bzip2"),
              (b"\xfd7zXZ\x00", "application/x-xz"), (b"\x1f\x9d", "application/x-compress"),
              (b"\x28\xb5\x2f\xfd", "application/zstd"), (b"LZIP", "application/x-lzip"),
              (b"PK\x03\x04", "application/zip"), (b"%PDF-", "application/pdf"))
drawn = ("text/", "application/xml", "application/json", "application/javascript", "application/pdf")

def from_bytes(head):
    for sig, t in signatures:
        if head.startswith(sig):
            return t
    text = b"\0" not in head or head.startswith((b"\xfe\xff", b"\xff\xfe"))
    return "text/plain" if text else "application/octet-stream"

def label(name, head):
    for ending, t in endings.items():
        if name.endswith(ending):
            return t
    t, encoding = mimetypes.guess_type("./" + name)
    seen = from_bytes(head)
    if encoding:
        return "application/octet-stream" if seen == "text/plain" else seen
    if t is None:
        return seen
    if t.startswith(drawn) or t.endswith(("+xml", "+json")):
        agrees = b"%PDF-" in head if t == "application/pdf" else seen == "text/plain"
        return t if agrees else seen
    return t

out = sys.stdout.buffer
for d, _, files in os.walk(root):
    for f in files:
        p = os.path.join(d, f)
        with open(p, "rb") as h:
            head = h.read(1024)
        out.write(os.fsencode(os.path.relpath(p, root)) + b"\t" + label(f, head).encode() + b"\n")
PY
